#include <iostream>
#include <string>
#include <map>

using namespace std;

int main() {
    int n;
    cout << "Enter number of symbols: ";
    cin >> n;

    char data[100];
    double freq[200];
    int parent[200], left[200], right[200];
    string codes[100]; // To store codes for encoding

    for (int i = 0; i < 200; i++) {
        parent[i] = left[i] = right[i] = -1;
    }

    // Input characters and frequencies together
    for (int i = 0; i < n; i++) {
        cout << "Enter symbol " << i + 1 << " and its frequency: ";
        cin >> data[i] >> freq[i];
    }

    int size = n;
    while (true) {
        int min1 = -1, min2 = -1;
        for (int i = 0; i < size; i++) {
            if (parent[i] == -1) {
                if (min1 == -1 || freq[i] < freq[min1]) {
                    min2 = min1;
                    min1 = i;
                } else if (min2 == -1 || freq[i] < freq[min2]) {
                    min2 = i;
                }
            }
        }
        if (min2 == -1) break;
        freq[size] = freq[min1] + freq[min2];
        left[size] = min1;
        right[size] = min2;
        parent[min1] = size;
        parent[min2] = size;
        size++;
    }

    cout << "\nHuffman Codes Table:\n";
    for (int i = 0; i < n; i++) {
        string currentCode = "";
        int curr = i;
        while (parent[curr] != -1) {
            int p = parent[curr];
            currentCode = (left[p] == curr ? "0" : "1") + currentCode;
            curr = p;
        }
        codes[i] = currentCode;
        cout << data[i] << " : " << codes[i] << endl;
    }

    // New Section: Encoding a particular string
    // New Section: Encoding a particular string
    string inputStr;
    cout << "\nEnter the string to encode: ";
    cin >> inputStr;

    cout << "Encoded Output: ";
    for (int j = 0; j < inputStr.length(); j++) {
        char c = inputStr[j];
        for (int i = 0; i < n; i++) {
            if (data[i] == c) {
                // Here "codes" should be the string array you stored earlier
                cout << codes[i];
                break;
            }
        }
    }
    cout << endl;
}
