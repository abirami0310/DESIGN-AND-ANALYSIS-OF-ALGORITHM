#include <iostream>
using namespace std;

// Function to print solution
void printSolution(int **board, int n) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cout << board[i][j] << " ";
        }
        cout << endl;
    }
}

// Check if safe
bool isSafe(int **board, int n, int row, int col) {

    // Check left side of row
    for (int i = 0; i < col; i++)
        if (board[row][i])
            return false;

    // Upper diagonal
    for (int i = row, j = col; i >= 0 && j >= 0; i--, j--)
        if (board[i][j])
            return false;

    // Lower diagonal
    for (int i = row, j = col; i < n && j >= 0; i++, j--)
        if (board[i][j])
            return false;

    return true;
}

// Backtracking function
bool solveNQ(int **board, int n, int col) {

    if (col >= n)
        return true;

    for (int i = 0; i < n; i++) {

        if (isSafe(board, n, i, col)) {
            board[i][col] = 1;

            if (solveNQ(board, n, col + 1))
                return true;

            // BACKTRACK
            board[i][col] = 0;
        }
    }

    return false;
}

int main() {
    int n;
    cout << "Enter number of queens: ";
    cin >> n;

    // Dynamic allocation
    int **board = new int*[n];
    for (int i = 0; i < n; i++) {
        board[i] = new int[n];
        for (int j = 0; j < n; j++)
            board[i][j] = 0;
    }

    if (solveNQ(board, n, 0)) {
        cout << "\nSolution:\n";
        printSolution(board, n);
    } else {
        cout << "No solution exists";
    }

    return 0;
}
